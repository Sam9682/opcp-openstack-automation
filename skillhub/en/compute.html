Compute

Difficulty: Intermediate · Estimated time: 25 min

Compute instances are the workhorses of your infrastructure. This lesson covers instance creation, flavor and image validation, user-data injection, metadata, and status polling — with both the Python SDK and Terraform.

1. Creating Instances with the SDK

The ComputeManager validates flavors and images against the OVHcloud catalog, creates instances, and waits for them to reach ACTIVE status (default timeout: 300 s). You can use environment variables to set INSTANCE_FLAVOR & INSTANCE_IMAGE. You also have to create a ssh key named 'opcp-openstack-automation-ssh-key' (use "ssh-keygen -f opcp-openstack-automation-ssh-key") 


import sys
import os
# Add the parent directory to Python path to import openstack_sdk
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from openstack_sdk.auth_manager import ConnectionManager
from openstack_sdk.compute_manager import ComputeManager
from openstack_sdk.network_manager import NetworkManager
from config.models import InstanceSpec, AuthCredentials
from openstack_sdk.auth_manager import AuthenticationManager

# Environment variables are loaded automatically when
# OS_APPLICATION_CREDENTIAL_ID and OS_APPLICATION_CREDENTIAL_SECRET are set
auth_manager = AuthenticationManager()
credentials = auth_manager.load_credentials_from_env()

connection = auth_manager.authenticate(credentials)

compute = ComputeManager(connection)
network = NetworkManager(connection)

# Get the network ID by name instead of using the name directly
student_id = os.environ.get('STUDENT_ID', 'default-student')
network_name = student_id + "-app-network"
network_obj = network.get_network_by_name(network_name)
if not network_obj:
    # Try to find any network that contains "app-network" to be more flexible
    print(f"Network '{network_name}' not found. Looking for similar networks...")
    all_networks = network.list_networks()
    app_networks = [net for net in all_networks if "app-network" in net.name]
    if app_networks:
        network_obj = app_networks[0]  # Use the first one found
        print(f"Using network: {network_obj.name}")
    else:
        raise Exception(f"Network '{network_name}' not found and no app-network found")

# Check if SSH keypair exists, if not create it
key_name = "opcp-openstack-automation-ssh-key"
try:
    # Try to get the keypair to see if it exists
    keypair = connection.compute.get_keypair(key_name)
    print(f"Keypair '{key_name}' already exists")
except Exception:
    # KeyPair doesn't exist, so create it
    print(f"Creating keypair '{key_name}'...")
    try:
        # Read public key from file
        pub_key_path = "./opcp-openstack-automation-ssh-key.pub"
        if os.path.exists(pub_key_path):
            with open(pub_key_path, 'r') as f:
                public_key = f.read().strip()

            # Create the keypair
            keypair = connection.compute.create_keypair(
                name=key_name,
                public_key=public_key
            )
            print(f"Created keypair '{key_name}' successfully")
        else:
            print(f"Warning: Public key file '{pub_key_path}' not found. "
                  f"Instance will be created without SSH keypair.")
    except Exception as e:
        print(f"Warning: Could not create keypair '{key_name}': {e}. "
              f"Instance will be created without SSH keypair.")

specs = [
    InstanceSpec(
        name=str(os.environ.get('STUDENT_ID', 'default-student'))+"-web-server-1",
        flavor=str(os.environ.get('INSTANCE_FLAVOR', 'scale-1')),
        image=str(os.environ.get('INSTANCE_IMAGE', 'Debian 12 LVM OPCP')),
        key_name=key_name,
        network_ids=[network_obj.id],
        security_groups=[student_id+"-web-server-sg"],
        metadata={"role": "web", "env": "production"},
    )
]

instances = compute.create_compute_instances(specs)
for inst in instances:
    print(f"{inst.name}  ID={inst.id}  Status={inst.status}")

3. User Data Injection

Cloud-init scripts passed via user_data run on first boot. Use them to install packages, start services, or configure the instance. 

user_data_script = """#!/bin/bash
apt-get update
apt-get install -y nginx
systemctl enable nginx
systemctl start nginx
"""

spec = InstanceSpec(
    name=str(os.environ.get('STUDENT_ID', 'default-student')) +"-nginx-host",
    flavor=str(os.environ.get('INSTANCE_FLAVOR', 'scale-1')),
    image=str(os.environ.get('INSTANCE_IMAGE', 'Debian 12 LVM OPCP')),
    key_name="opcp-openstack-automation-ssh-key",
    network_ids=[str(os.environ.get('STUDENT_ID', 'default-student')) +"-private-network"],
    security_groups=[str(os.environ.get('STUDENT_ID', 'default-student')) +"-web-security-group"],
    user_data=user_data_script,
)

instances = compute.create_compute_instances([spec])

4. Metadata

Attach arbitrary key-value metadata to instances for tagging, cost allocation, or automation hooks. 

spec = InstanceSpec(

    },
)

5. Status Polling

After creation the manager polls the API every 5 seconds until the instance reaches ACTIVE or the 300-second timeout expires. If the instance enters ERROR state the fault details are reported immediately. 

# Wait for a specific instance to become ACTIVE
instance = compute.wait_for_instance_active("instance-id", timeout=300)
print(f"Instance is now {instance.status}")

# Query instances
inst = compute.get_instance_by_name(str(os.environ.get('STUDENT_ID', 'default-student')) +"-web-server-1")
all_instances = compute.list_instances()

# Delete when done
compute.delete_instance("instance-id")

6. Compute with Terraform

resource "openstack_compute_instance_v2" "instance" {
  count       = var.instance_count
  name        = "${var.network_name}-instance-${count.index + 1}"
  flavor_name = var.instance_flavor
  image_name  = var.instance_image
  key_pair    = var.key_name

  network {
    uuid = openstack_networking_network_v2.private_network.id
  }

  metadata  = var.instance_metadata
  user_data = var.instance_user_data

  depends_on = [
    openstack_networking_subnet_v2.private_subnet
  ]
}
# Apply the Terraform configuration
cd terraform
terraform init
terraform plan -var="instance_count=2"
terraform apply -auto-approve

Summary

  • ComputeManager validates flavors and images before creation.
  • Inject cloud-init scripts via user_data for first-boot automation.
  • Attach metadata for tagging and cost tracking.
  • The SDK polls until ACTIVE (300 s timeout, 5 s interval).
  • Terraform provides a declarative alternative with count-based scaling.
